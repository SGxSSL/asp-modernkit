function placeFocus() {
if (document.forms.length > 0) {
	var field = document.forms[0];
	for (i = 0; i < field.length; i++) {
		var input = field.elements[i]
		if ((input.type == "text") || (input.type == "textarea") ||	(input.type == "password")) {
			if (input.value == "") {
				input.focus();
				break;
			}
		} else if (input.type == "submit") {
			input.focus();
			break;
		}
	}
}
return true;
}
